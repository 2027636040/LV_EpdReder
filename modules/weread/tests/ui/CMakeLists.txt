cmake_minimum_required(VERSION 3.20)
project(weread_ui_tests C)
set(CMAKE_C_STANDARD 11)
get_filename_component(REPO "${CMAKE_CURRENT_LIST_DIR}/../../../.." ABSOLUTE)
set(LVGL "${REPO}/SiFli-SDK/external/lvgl_v9")
file(GLOB_RECURSE LV_SOURCES CONFIGURE_DEPENDS "${LVGL}/src/*.c")
add_library(ui_lvgl STATIC ${LV_SOURCES})
target_include_directories(ui_lvgl PUBLIC "${LVGL}" "${CMAKE_CURRENT_LIST_DIR}/../ui_include")
target_compile_definitions(ui_lvgl PUBLIC LV_KCONFIG_IGNORE LV_CONF_INCLUDE_SIMPLE)
target_compile_options(ui_lvgl PUBLIC -g -O1 -fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
add_executable(ui_test ../ui_test.c)
target_compile_options(ui_test PRIVATE -Wall -Wextra -Werror)
target_link_options(ui_test PRIVATE -fsanitize=address,undefined -no-pie)
target_link_libraries(ui_test PRIVATE ui_lvgl m)
