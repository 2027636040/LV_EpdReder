cmake_minimum_required(VERSION 3.20)
project(reader_public_tests C)
set(CMAKE_C_STANDARD 11)
get_filename_component(REPO "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(LVGL "${REPO}/SiFli-SDK/external/lvgl_v9")
enable_testing()

# Reuse the lifecycle test's RTOS substitutes with unmodified function bodies.
file(READ "${REPO}/src/services/app_service.c" SERVICE)
string(REGEX REPLACE "#include[^\n]*\n" "" SERVICE "${SERVICE}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/service_under_test.h" "${SERVICE}")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
    "${REPO}/src/services/app_service.c")
foreach(extension h c)
    file(READ "${REPO}/src/ui/platform/epd_app.${extension}" API)
    string(FIND "${API}" "const lv_draw_buf_t *epd_app_font_derived_acquire(" START)
    if(extension STREQUAL "h")
        string(FIND "${API}" "/* UI-thread operations. Libraries" END)
    else()
        string(FIND "${API}" "RTM_EXPORT(lv_draw_wait_for_finish);" END)
    endif()
    math(EXPR LENGTH "${END} - ${START}")
    string(SUBSTRING "${API}" ${START} ${LENGTH} API)
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/derived_api_${extension}.h" "${API}")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
        "${REPO}/src/ui/platform/epd_app.${extension}")
endforeach()
foreach(rate 1000 100)
    add_executable(ui_deadline_${rate} test_ui_deadline.c)
    target_include_directories(ui_deadline_${rate} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
    target_compile_definitions(ui_deadline_${rate} PRIVATE RT_TICK_PER_SECOND=${rate}
        _CRT_SECURE_NO_WARNINGS)
    if(MSVC)
        target_compile_options(ui_deadline_${rate} PRIVATE /W4 /WX /utf-8 /UNDEBUG)
    else()
        target_compile_options(ui_deadline_${rate} PRIVATE -Wall -Wextra -Werror -UNDEBUG)
    endif()
    add_test(NAME ui_deadline_${rate} COMMAND ui_deadline_${rate})
endforeach()

# Only the host LVGL library is built, with the real pixel LRU and draw buffers.
file(GLOB_RECURSE LV_SOURCES CONFIGURE_DEPENDS "${LVGL}/src/*.c")
add_library(reader_lvgl STATIC ${LV_SOURCES})
target_include_directories(reader_lvgl PUBLIC "${LVGL}" "${CMAKE_CURRENT_LIST_DIR}/include")
target_compile_definitions(reader_lvgl PUBLIC LV_KCONFIG_IGNORE LV_CONF_INCLUDE_SIMPLE
    _CRT_SECURE_NO_WARNINGS)
add_executable(derived_cache test_derived_cache.c)
target_include_directories(derived_cache PRIVATE "${REPO}/src/ui" "${REPO}/src/platform"
    "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(derived_cache PRIVATE reader_lvgl)
if(MSVC)
    target_compile_options(reader_lvgl PRIVATE /utf-8 /MP)
    target_compile_options(derived_cache PRIVATE /W4 /utf-8 /UNDEBUG)
else()
    target_compile_options(derived_cache PRIVATE -Wall -Wextra -UNDEBUG)
    target_link_libraries(derived_cache PRIVATE m)
endif()
add_test(NAME derived_cache COMMAND derived_cache)
